\section{Navigation Layer}
\label{sec:navigation}

The navigation layer builds on top of the peer-sampling layer, building a structure that prioritizes the establishment of connections between nodes interested in similar topics. As specified in \cite{akv24}, the chosen peer-to-peer scheme for the navigation layer is Vicinity \cite{vs13}.

\begin{algorithm}[H]
\caption{\navsetup}
\label{alg:navsetup}
\textbf{Input:}
\begin{itemize}
\vspace{-.8em}
\itemsep0em 
    \item $1^\kappa$ The security parameter.
    \item $\aux$ Any additional auxiliary data.
\end{itemize}
\textbf{Output:}
\begin{itemize}
\vspace{-.8em}
\itemsep0em 
    \item $\state$ Updated node state.
\end{itemize}
\begin{algorithmic}
    \State $(\routingbase,\nodesperfinger) \gets \aux$
    \State $\state.\navigation \gets (\routingbase,\nodesperfinger,\emptyset)$
    \State \return $\state$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{\navselect. Navigation selection algorithm, approximating the caller's outgoing links in the target structure}
\label{alg:navselect}
\textbf{Input:}
\begin{itemize}
\vspace{-.8em}
\itemsep0em 
    \item $\D$ The reference node's descriptor (may not be the caller.)
    \item $\Dset_{in}$ A set of node descriptors.
    \item $k$ The number of node descriptors to select per finger.
\end{itemize}
\textbf{Output:}
\begin{itemize}
\vspace{-.8em}
\itemsep0em 
    \item $\Dset_{out}$ The calling peer's new (proposed) view.
\end{itemize}
\begin{algorithmic}
    \State $(\routingbase,\cdot,\cdot) \gets \state.\navigation$
    %\State $OT \gets OrderByTopic(\topicReg)$
    \State $\Dset_{out} \gets \emptyset$
    \State \codecomment{Get topics that $\D$ is subscribed to \todo{How is this done?}}
    \State $\topicIds \gets GetSubscriptions(\D)$
    \State \codecomment{For each subscribed topic, iterate through its $b^i$ distant finger topics}
    \For{$t \in \topicIds$}
    \State $\Dset_{out} \getscup~FindClosestFreshestPeersPerTopic(\Dset_{in},t,k)$
    \For{$i\in [0,\floor{\log_b(|\topicReg|/2)}+1]$}
    \State \codecomment{Find $c$ subscribers as close to $\topicId$ $t \pm b^i$ as possible}
    \State $\Dset_{out} \getscup~FindClosestFreshestPeersPerTopic(\Dset_{in},t+b^i,k)$
    \State $\Dset_{out} \getscup~FindClosestFreshestPeersPerTopic(\Dset_{in},t-b^i,k)$
    \EndFor
    \EndFor
    \State \Return $\Dset_{out}$ \codecomment{$\Dset_{out}$ has size $(2*(\floor{\log_b(|\topicReg|/2)}+1)+1)*k*|\topicIds|$}
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{\navsend}
\label{alg:navsend}
\textbf{Input:}
\begin{itemize}
\vspace{-.8em}
\itemsep0em 
    \item $\D$ the target descriptor.
\end{itemize}
\textbf{Output:}
\begin{itemize}
\vspace{-.8em}
\itemsep0em 
    \item $\top$ if ok, $\bot$ otherwise.
\end{itemize}
\begin{algorithmic}
    \State $(\cdot,\cdot,\view) \gets \state.\navigation$
    \State $\Dset \gets \navselect(\D,\view,1)$ \codecomment{\todo{\jesus{Should $\view$ here be set to the $\peersampling$ view? If not, how and when do we feed $\peersampling$ nodes into $\navigation$?}}}
    \State $\network.\send(\D.\netaddr,\Dset)$
    \State \Return $\top$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{\navrecv}
\label{alg:navrecv}
\textbf{Input:}
\begin{itemize}
\vspace{-.8em}
\itemsep0em 
    \item $\D$ the descriptor of the calling party.
    \item $\Dset_{in}$ the received set of descriptors (sent by some other party).
\end{itemize}
\textbf{Output:}
\begin{itemize}
\vspace{-.8em}
\itemsep0em 
    \item $\top$ if ok, $\bot$ otherwise.
\end{itemize}
\begin{algorithmic}
    \State $(\cdot,\nodesperfinger,\view) \gets \state.\navigation$
    \State $\Dset_{out} \gets \navselect(\D,\view\cup\Dset_{in},c)$
    \State $\state.\navigation.\view \gets \Dset_{out}$
    \State \Return $\Dset_{out}$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{\navrun}
\label{alg:navrun}
\textbf{Input:}
\begin{itemize}
\vspace{-.8em}
\itemsep0em 
    \item $\routingbase$, the routing base.
    \item $\nodesperfinger$, the number of nodes per finger.
    %\item $\D_{btp}$, a bootstrapping node (may be extended to a set)
\end{itemize}
\textbf{Output:}
\begin{itemize}
\vspace{-.8em}
\itemsep0em 
    \item Nothing. The program runs until stopped.
\end{itemize}
\begin{algorithmic}
\State $\state \gets \navsetup(\routingbase,\nodesperfinger)$
\While{\textbf{true}}
\State \codecomment{Receive and send can run concurrently (but view must be mutexed?)}
\State $($
\State ~$((\D,\Dset) \gets \network.\receive(\ast);$ $\navrecv(\D, \Dset))$
\State ~$|$
%
\State ~$(\D_{snd} \getsr \state.\navigation.\view;$ $\navsend(\D_{snd}))$
\State $)$
\EndWhile
\end{algorithmic}
\end{algorithm}
